## What it does

Curve/key lookup with binary search + lerp clamp. Struct arg (`a0`):
`f32 q @0x00`, `s32 type @0x04` (must be 16), `s16 *keys @0x08`, `s32 base @0x0C`,
`s32 count @0x10`, `s32 i1 @0x14`, `s32 i2 @0x18`, `s32 k1 @0x1C`, `s32 k2 @0x20`.
Float globals (gp-relative): `D_003DF3BC` (lower threshold), `D_003DF40C` (upper).

Shape: `count`, `lo=0`, `k1=k2=0`, `q`, `keys`, `qi=(s32)q` in that emission order
(orig regs: hi=$t1, lo=$t2, q=$f2, keys=$v0, qi=$v1, sum=$a3, &k1=$a2, &k2=$a1).
`if (keys != 0)`: type!=16 → `i2=0` in the `bnel` delay slot, out-of-line block sets
f0=0, `i1=0`, return. Binary search `while (lo < hi) { mid=(lo+hi)/2; if (qi<keys[mid])
hi=mid; else lo=mid+1; }` — signed `/2` via plain `bgez`+`sra` with the `addiu +1`
fix-up (SKILL/idioms signed-div-by-2 pattern), sum `lo+hi` computed in the back-edge
`bnel` delay slot. Then `i1=lo-1; i2=lo`, reload `i2` via field, several guard
returns 0 (idiom "beql delay slot loads the store/result, EE discards on not-taken":
`lw $v0,0x14($a0)` and `addiu $v0,$a3,-1` in beql delay slots, `lw $v0,0x18($a0)` in
the `bc1fl` delay slot). Then `k1=keys[i1]; k2=keys[i2]` via `a2`/`a1` pointer locals,
stored, **reloaded from the fields** (`lw 0x1C(a0)`, `lw 0x20(a0)`) for
`t=(f32)(k2-k1)`; `if (t>0)` (c.le.s f0,zero) `t=(q-(f32)c->k1)/t`. `else` (keys==0)
out-of-line at 0x156B5C: `d=qi-c->base; c->i1=d; c1=c->count-1;` guards
`i1>=c1` → both= c1 return 0; `i2=i1+1; i2<=0` → i1=0; `i2>=count` → `i2=i1`;
then `k1=qi; k2=qi+1; t=q-(f32)qi;` b back to the shared clamp: `if (t<D_003DF3BC)
return 0; if (t<=D_003DF40C) return t; i1=i2; k1=k2; return 0;`.
Orig block order: main inline; out-of-line: null-keys path (B5C), then cold
type-block (C0C), clamp-low (C18), clamp-high (C24), epilogue `jr $ra; nop` (C34).

```c
// range: 0x156A20 0x156C40
#include "common.h"

struct Curve {
    f32 q;       /* +0x00 */
    s32 type;    /* +0x04 */
    s16 *keys;   /* +0x08 */
    s32 base;    /* +0x0C */
    s32 count;   /* +0x10 */
    s32 i1;      /* +0x14 */
    s32 i2;      /* +0x18 */
    s32 k1;      /* +0x1C */
    s32 k2;      /* +0x20 */
};

extern f32 D_003DF3BC;
extern f32 D_003DF40C;

f32 func_00156A20(struct Curve *c) {
    s32 hi, lo, qi, d, c1, mid;
    s16 *keys;
    f32 q, t;

    hi = c->count;
    lo = 0;
    c->k1 = 0;
    c->k2 = 0;
    q = c->q;
    keys = c->keys;
    qi = (s32)q;
    if (keys != 0) {
        if (c->type != 16) {
            c->i2 = 0;
            c->i1 = 0;
            return 0.0f;
        }
        while (lo < hi) {
            mid = (lo + hi) / 2;
            if (qi < keys[mid]) {
                hi = mid;
            } else {
                lo = mid + 1;
            }
        }
        c->i1 = lo - 1;
        c->i2 = lo;
        if (c->i2 == 0) {
            c->i1 = 0;
            return 0.0f;
        }
        hi = c->count;
        if (c->i2 >= hi) {
            c->i2 = c->i1;
            return 0.0f;
        }
        if (c->i1 >= hi) {
            c->i2 = hi - 1;
            c->i1 = hi - 1;
            return 0.0f;
        }
        c->k1 = keys[c->i1];
        c->k2 = keys[c->i2];
        t = (f32)(c->k2 - c->k1);
        if (t > 0.0f) {
            t = (q - (f32)c->k1) / t;
        }
    } else {
        d = qi - c->base;
        c->i1 = d;
        c1 = c->count - 1;
        if (c->i1 >= c1) {
            c->i2 = c1;
            c->i1 = c1;
            return 0.0f;
        }
        c->i2 = c->i1 + 1;
        if (c->i2 <= 0) {
            c->i1 = 0;
            return 0.0f;
        }
        if (c->i2 >= c->count) {
            c->i2 = c->i1;
            return 0.0f;
        }
        c->k1 = qi;
        c->k2 = qi + 1;
        t = q - (f32)qi;
    }
    if (t < D_003DF3BC) {
        return 0.0f;
    }
    if (t <= D_003DF40C) {
        return t;
    }
    c->i1 = c->i2;
    c->k1 = c->k2;
    return 0.0f;
}
```
