// copy of the free space, then byte by byte.  With the buffer full the byte

// handler stops the write.  Returns the number of bytes written.

    u8 *end;                        /* +0x18 */
};

s64 func_00103980(struct Writer *o, s8 *src, s64 n) {
    s64 space = o->end - o->cursor;
    s64 *pick;
    s64 written;
    s32 ch;
    s32 result;

    if (space < n) {
        pick = &space;
    } else {
        pick = &n;

    written = *pick;
    if (written > 0) {
        func_00130FD0(o->cursor, src, (s32)written);
        o->cursor += (s32)written;
        src += (s32)written;
        n -= written;

    while (n > 0) {
        ch = *src++;
        if (o->cursor < o->end) {
