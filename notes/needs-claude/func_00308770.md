#include "common.h"

s32 func_00308770(void *arg0, s32 arg1) {
    u32 *bitmap;
    u32 word;
    s32 bit;
    s32 mask;

    if (arg1 < 0 || arg1 >= 0x800) {
        return -1;
    }
    bitmap = (u32 *)((char *)arg0 + 0x440);

    bit = arg1 % 32;
    mask = 1 << bit;
    return (word & mask) != 0;
}
