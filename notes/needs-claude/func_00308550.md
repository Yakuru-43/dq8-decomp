#include "common.h"

struct BitMap {
    char pad[0x50];
    u32 words[0x40];
};

s32 func_00308550(struct BitMap *map, s32 bit, s32 set) {
    s32 valid;
    s32 old;
    s32 mask;

    if (bit >= 0 && bit < 0x800) {
        valid = 1;
    } else {
        valid = 0;
    }
    if (valid != 0) {
        mask = 1 << (bit % 32);
        old = (map->words[bit / 32] & mask) != 0;
        if (set) {
            map->words[bit / 32] |= mask;
        } else {
            map->words[bit / 32] &= ~mask;
        }
    } else {
        old = 0;
    }
    return old;
}
