## What it does
Under `WaitSema(page->sema)` (sema at 0x40, same guard as `func_0019F820`):
computes `rem = (((m0C + m10) << 11) + m14) % m18` and
`total = ((m08 - 2 - m10) << 11) - m14`; then if `m18 - rem < total` writes
four outputs: `*o0 = m00 + rem, *o1 = m18 - rem, *o2 = m00,
*o3 = total - (m18 - rem)`; else `*o0 = m00 + rem, *o1 = total, *o2 = 0,
*o3 = 0`. Then `SignalSema`. Five pointer args (5th in `$t0` → `$s0` in the
`WaitSema` delay slot).

```c
// range: 0x19FC10 0x19FD10
// flags: -flag divbyzerocheck
#include "common.h"

M2C_UNK WaitSema(s32);                            /* extern */
M2C_UNK SignalSema(s32);                          /* extern */

typedef struct {
    /* 0x00 */ s32 m00;
    /* 0x04 */ s32 m04;
    /* 0x08 */ s32 m08;
    /* 0x0C */ s32 m0C;
    /* 0x10 */ s32 m10;
    /* 0x14 */ s32 m14;
    /* 0x18 */ s32 m18;
    /* 0x1C */ u8 pad1C[0x24];
    /* 0x40 */ s32 sema;
} VuPage;

void func_0019FC10(VuPage *page, s32 *out0, s32 *out1, s32 *out2, s32 *out3) {
    s32 total;
    s32 rem;

    WaitSema(page->sema);
    rem = (((page->m0C + page->m10) << 11) + page->m14) % page->m18;
    total = ((page->m08 - 2 - page->m10) << 11) - page->m14;
    if (!(page->m18 - rem < total)) {
        *out0 = page->m00 + rem;
        *out1 = total;
        *out2 = 0;
        *out3 = 0;
    } else {
        *out0 = page->m00 + rem;
        *out1 = page->m18 - rem;
        *out2 = page->m00;
        *out3 = total - (page->m18 - rem);
    }
    SignalSema(page->sema);
}
```
